class LoginSerializer(serializers.Serializer):
    email = serializers.EmailField()
    password = serializers.CharField(write_only=True)

    def validate(self, data):
        user = authenticate(
            request=self.context.get("request"),
            email=data["email"],
            password=data["password"],
        )

        if user and user.is_active:
            return user
        raise serializers.ValidationError("Invalid credentials")

class RiderSignupSerializer(serializers.ModelSerializer):
    email = serializers.EmailField(write_only=True)
    first_name = serializers.CharField(write_only=True)
    last_name = serializers.CharField(write_only=True)
    password = serializers.CharField(
        write_only=True, style={"input_type": "password"})

    class Meta:
        model = Rider
        fields = [
            "email",
            "first_name",
            "last_name",
            "password",
            "vehicle_type",
            "license_number",
            "contact_number",
            "address",
        ]

    def create(self, validated_data):
        email = validated_data.pop("email")
        first_name = validated_data.pop("first_name")
        last_name = validated_data.pop("last_name")
        password = validated_data.pop("password")
        user = CustomUser.objects.create(
            email=email, first_name=first_name, last_name=last_name,
            is_rider=True,
            # is_staff=True,
            is_owner=validated_data.get("is_owner", False),
        )
        user.set_password(password)
        user.save()
        rider = Rider.objects.create(user=user, **validated_data)

        rider_wallet = RiderWallet.objects.create(rider=rider)

        rider_wallet.add_funds(0)

        return rider

class LoginViewSet(viewsets.ViewSet):
    permission_classes = [AllowAny]
    serializer_class = LoginSerializer

    def create(self, request):
        serializer = self.serializer_class(data=request.data)
        if serializer.is_valid():
            user = serializer.validated_data
            refresh = RefreshToken.for_user(user)
            access_token = str(refresh.access_token)
            token, created = Token.objects.update_or_create(user=user)

            return Response({
                'access': access_token,
                'refresh': str(refresh),
                'token': token.key
            }, status=status.HTTP_200_OK)

        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

class CustomUserManager(BaseUserManager):
    def create_user(self, email, password=None, **extra_fields):
        if not email:
            raise ValueError('A valid email address must be provided')
        email = self.normalize_email(email)
        user = self.model(email=email, **extra_fields)
        user.set_password(password)
        user.save(using=self._db)
        return user

    def create_superuser(self, email, password=None, **extra_fields):
        extra_fields.setdefault('is_staff', True)
        extra_fields.setdefault('is_superuser', True)

        if extra_fields.get('is_staff') is not True:
            raise ValueError('Superuser must have is_staff=True.')
        if extra_fields.get('is_superuser') is not True:
            raise ValueError('Superuser must have is_superuser=True.')

        return self.create_user(email, password, **extra_fields)


class CustomUser(AbstractBaseUser, PermissionsMixin):
    groups = models.ManyToManyField(
        'auth.Group',
        related_name='customuser_groups',  # Change related_name
        blank=True
    )
    user_permissions = models.ManyToManyField(
        'auth.Permission',
        related_name='customuser_permissions',  # Change related_name
        blank=True
    )
    email = models.EmailField(unique=True)
    first_name = models.CharField(max_length=30)
    last_name = models.CharField(max_length=30)
    is_owner = models.BooleanField(default=False)
    is_active = models.BooleanField(default=True)
    is_staff = models.BooleanField(default=False)
    is_rider = models.BooleanField(default=False)

    USERNAME_FIELD = 'email'
    REQUIRED_FIELDS = ['first_name', 'last_name']

    objects = CustomUserManager()

    class Meta:
        verbose_name = 'User'
        verbose_name_plural = 'Users'

    def __str__(self):
        return f"{self.first_name} {self.last_name}"

    def update_email(self, new_email):
        if CustomUser.objects.filter(email=new_email).exclude(id=self.id).exists():
            raise ValueError('This email is already in use.')
        self.email = new_email
        self.save(update_fields=['email'])

class Customer(models.Model):
    user = models.OneToOneField(
        CustomUser,
        on_delete=models.CASCADE,
        related_name='customer_profile'
    )
    phone_number = models.CharField(max_length=15, blank=True, null=True)
    address = models.TextField(blank=True, null=True)
    address_latitude = models.DecimalField(
        max_digits=9, decimal_places=6, null=True, blank=True
    )
    address_longitude = models.DecimalField(
        max_digits=9, decimal_places=6, null=True, blank=True
    )
    created_at = models.DateTimeField(auto_now_add=True, null=True)
    updated_at = models.DateTimeField(auto_now=True, null=True)

    def __str__(self):
        return f"{self.user.first_name} {self.user.last_name} - Customer"

    def save(self, *args, **kwargs):
        # Ensure is_staff is set to True when the customer is created
        if not self.user.is_staff:
            self.user.is_staff = True
        super().save(*args, **kwargs)

    def get_full_address(self):
        # Example of combining address fields
        return f"{self.address}, Lat: {self.address_latitude}, Lon: {self.address_longitude}"

    @property
    def full_name(self):
        return f"{self.user.first_name} {self.user.last_name}"

    def has_valid_address(self):
        """Check if customer has a valid address."""
        return bool(self.address and self.address_latitude and self.address_longitude)

    def update_location(self, latitude, longitude):
        """Update address location information."""
        self.address_latitude = latitude
        self.address_longitude = longitude
        self.save()

def assign_rider_to_order(order):
    """Assign the nearest rider to an order."""
    riders = Rider.objects.filter(is_available=True).exclude(
        latitude=None, longitude=None)
    nearest_rider = None
    shortest_distance = float('inf')
    for rider in riders:
        if rider.latitude and rider.longitude:
            distance = calculate_distance(
                order.delivery_latitude, order.delivery_longitude,
                rider.latitude, rider.longitude, order.delivery_longitude,
                float(rider.latitude), float(rider.longitude)
            )
            if distance < shortest_distance:
                shortest_distance = distance
                nearest_rider = rider
                nearest_rider = rider
    if nearest_rider:
        order.rider = nearest_rider
        order.save()
        nearest_rider.is_available = False
        nearest_rider.save()

class ManualOrderSerializer(serializers.ModelSerializer):
    customer_name = serializers.SerializerMethodField()
    product_image = serializers.ImageField(required=False, allow_null=True)
    invoice = serializers.ImageField(required=False, allow_null=True)
    time_remaining = serializers.SerializerMethodField()
    customer_mobile = serializers.SerializerMethodField()

    class Meta:
        model = ManualOrder
        fields = [
            'id', 'customer', 'customer_name', 'name', 'mobile_number', 'address', 'product_name',
            'product_image', 'quantity', 'unit', 'order_details', 'total_price',
            'ride_charges', 'paid_amount', 'status', 'rider', 'invoice', 'created_at',
            'updated_at', 'time_remaining', 'customer_mobile'
        ]
        read_only_fields = ['id', 'created_at', 'updated_at', 'user', 'rider']
    def create(self, validated_data):
        # Ensure that the customer is being passed correctly and associated with the order
        customer = validated_data.pop('customer')  # You can also set a default if needed
        order = ManualOrder.objects.create(customer=customer, **validated_data)
        return order
    def get_customer_name(self, obj):
        """Retrieve the customer's full name."""
        if obj.customer and obj.customer.user:
            return f"{obj.customer.user.first_name} {obj.customer.user.last_name}"
        return None

    def get_time_remaining(self, obj):
        elapsed_time = timezone.now() - obj.created_at
        remaining_time = timedelta(minutes=20) - elapsed_time

        if remaining_time.total_seconds() > 0:
            return remaining_time.total_seconds()
        else:
            return 0

    def get_customer_mobile(self, obj):
        """Return mobile number if the user is an admin or the customer who placed the order."""
        request = self.context.get('request')
        if request and request.user.is_staff:
            return obj.mobile_number
        elif request and request.user == obj.customer:
            return obj.mobile_number
        return None

    def __init__(self, *args, **kwargs):
        # Pop the request from the context
        request = kwargs['context'].get(
            'request') if 'context' in kwargs else None
        super().__init__(*args, **kwargs)

        if request and not (request.user.is_staff or request.user.is_superuser):
            self.fields.pop('total_price')
            self.fields.pop('ride_charges')
            self.fields.pop('paid_amount')
            self.fields.pop('invoice')
            self.fields.pop('status')

        if request and not request.user.is_staff:
            self.fields.pop('mobile_number')

    def validate(self, data):
        if not data.get('address'):
            raise serializers.ValidationError("address is required.")
        if not data.get('product_name'):
            raise serializers.ValidationError("Product name is required.")
        if 'quantity' in data and data.get('quantity') <= 0:
            raise serializers.ValidationError(
                "Quantity must be greater than 0.")
        if 'total_price' in data and data.get('total_price') <= 0:
            raise serializers.ValidationError(
                "Total price must be greater than 0.")
        if 'paid_amount' in data and 'total_price' in data:
            total_price = data.get('total_price', 0)
            paid_amount = data.get('paid_amount', 0)
            if paid_amount < total_price:
                raise serializers.ValidationError(
                    "Paid amount must be greater than or equal to the total price.")
        return data

class ManualOrderViewSet(viewsets.ModelViewSet):
    queryset = ManualOrder.objects.all()
    serializer_class = ManualOrderSerializer
    permission_classes = [IsAuthenticated]
    authentication_classes = [SessionAuthentication, JWTAuthentication]

    def list(self, request, *args, **kwargs):
        user = self.request.user
        if user.is_superuser:
            # Separate completed and active orders
            completed_orders = ManualOrder.objects.filter(status='delivered')
            active_orders = ManualOrder.objects.exclude(
                Q(status='delivered') | Q(rider__isnull=True))
            unassigned_orders = ManualOrder.objects.filter(
                Q(rider=user) | Q(rider__isnull=True))
        elif user.is_staff:
            # Riders see only orders assigned to them
            completed_orders = ManualOrder.objects.filter(
                rider=user, status='delivered')
            active_orders = ManualOrder.objects.filter(
                rider=user).exclude(status='delivered')
            unassigned_orders = None
        else:
            # Normal users see only their own orders
            completed_orders = ManualOrder.objects.filter(
                user=user, status='delivered')
            active_orders = ManualOrder.objects.filter(
                user=user).exclude(status='delivered')
            unassigned_orders = None

        # Serialize the querysets
        completed_orders = self.get_serializer(
            completed_orders, many=True).data
        active_orders = self.get_serializer(active_orders, many=True).data
        unassigned_orders = self.get_serializer(
            unassigned_orders, many=True).data if unassigned_orders else []

        return Response({
            "completed_orders": completed_orders,
            "active_orders": active_orders,
            'unassigned_orders': unassigned_orders,
        })

    def perform_create(self, serializer):
        user = self.request.user

        if not user.is_superuser:
            # Get the order address from the serializer's validated data
            order_address = serializer.validated_data.get('address')

            # Check for available rider based on address
            rider = User.objects.filter(
                Q(is_superuser=False) & Q(is_staff=True))
            rider = rider.filter(profile__address=order_address).first()

            if rider:
                serializer.save(user=user, rider=rider)
            else:
                rider = User.objects.filter(
                    Q(is_superuser=True) & Q(is_staff=False)).first()
                if rider:
                    serializer.save(user=user, rider=rider)
                else:
                    return Response(
                        {'detail': 'No available riders for the specified address, Order has been shifted to unassigned orders'},
                        status=status.HTTP_400_BAD_REQUEST
                    )
        else:
            serializer.save(user=user)

    def perform_update(self, serializer):
        user = self.request.user
        instance = serializer.instance

        if user.is_staff and not user.is_superuser:
            # Staff users can only update price, status, and invoice fields
            serializer.save(
                total_price=serializer.validated_data.get(
                    'total_price', instance.total_price),
                status=serializer.validated_data.get(
                    'status', instance.status),
                invoice=serializer.validated_data.get(
                    'invoice', instance.invoice),
                rider=instance.rider,  # Preserve rider
                user=instance.user  # Preserve user
            )
        elif user.is_superuser:
            # Superuser can change all fields
            serializer.save()
        else:
            # If user is not permitted to update this order, deny access
            return Response(
                {'detail': 'You do not have permission to update this order.'},
                status=status.HTTP_403_FORBIDDEN
            )


@api_view(['GET'])
@permission_classes([IsAuthenticated])
def complete_order(request, order_id):
    if not request.user.is_staff:
        return Response({"error": "Only admin and rider users can complete orders."}, status=status.HTTP_403_FORBIDDEN)
    try:
        order = ManualOrder.objects.get(id=order_id)

        # Convert order values to Decimal to avoid float issues
        ride_charges = Decimal(order.ride_charges)
        paid_amount = Decimal(order.paid_amount)
        total_price = Decimal(order.total_price)

        total_required = total_price + ride_charges

        if paid_amount > total_required:
            excess_amount = paid_amount - total_required
            customer_wallet, _ = CustomerWallet.objects.get_or_create(
                customer=order.user)
            customer_wallet.add_funds(excess_amount)

        elif paid_amount < total_required:
            deficit_amount = total_required - paid_amount
            customer_wallet, _ = CustomerWallet.objects.get_or_create(
                customer=order.user)

            if customer_wallet.balance >= deficit_amount:
                customer_wallet.deduct_funds(deficit_amount)
            else:
                return Response({"error": "Insufficient funds in customer's wallet to cover the deficit."},
                                status=status.HTTP_400_BAD_REQUEST)

        # Distribution Logic
        company_share_percentage = Decimal('0.07')
        company_share = order.ride_charges * company_share_percentage
        rider_share = order.ride_charges - company_share

        # Ensure Company Wallet exists or create it
        company_wallet, created = CompanyWallet.objects.get_or_create(id=1)
        company_wallet.add_funds(company_share)

        # Update the rider's wallet or create if not available
        rider_wallet, _ = RiderWallet.objects.get_or_create(rider=order.rider)
        rider_wallet.add_funds(rider_share)

        # Mark the order as completed (delivered)
        order.status = 'delivered'
        order.save()

        return Response({"message": "Order completed and wallets updated successfully."}, status=status.HTTP_200_OK)
    except ManualOrder.DoesNotExist:
        return Response({"error": "Order not found."}, status=status.HTTP_404_NOT_FOUND)
    except Exception as e:
        return Response({"error": str(e)}, status=status.HTTP_400_BAD_REQUEST)

class CreateManualOrder(APIView):
    permission_classes = [IsAuthenticated]

    def post(self, request, *args, **kwargs):
        if not request.user.is_authenticated:
            raise ValidationError("Authentication is required to create an order.")

        request_data = request.data
        request_data['user'] = request.user.id

        try:
            validCustomer = Customer.objects.get(user=request.user)
        except Customer.DoesNotExist:
            raise NotFound("Customer matching query does not exist.")

        # Get the customer's latitude and longitude
        customer_lat = validCustomer.address_latitude
        customer_lon = validCustomer.address_longitude

        # Check if the customer has valid location data
        if not customer_lat or not customer_lon:
            raise ValidationError("Customer must have a valid address with latitude and longitude.")

        # Find the nearest rider
        nearest_rider = None
        min_distance = float('inf')  # Start with a very large number

        riders = Rider.objects.all()  # Get all riders (you can filter based on availability if needed)
        for rider in riders:
            if rider.latitude and rider.longitude:
                distance = calculate_distance(customer_lat, customer_lon, rider.latitude, rider.longitude)
                if distance < min_distance:
                    min_distance = distance
                    nearest_rider = rider

        if not nearest_rider:
            raise NotFound("No available riders.")

        # Add the nearest rider to the request data
        request_data['rider'] = nearest_rider.id  # Assign the nearest rider's ID

        # Serialize the data using ManualOrderSerializer
        serializer = ManualOrderSerializer(data=request_data, context={'request': request})

        if serializer.is_valid():
            manual_order = serializer.save(customer=validCustomer, rider=nearest_rider)

            return Response(
                {"message": "Order created successfully.", "order": serializer.data},
                status=status.HTTP_201_CREATED,
            )

        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)
